import json, glob
from collections import defaultdict
import math

F = ["dialog", "dialog+query"]

def wilson(k, n, z=1.96):
    if not n: return (0, 0)
    p = k/n; d = 1 + z*z/n
    c = (p + z*z/(2*n))/d
    h = z*math.sqrt(p*(1-p)/n + z*z/(4*n*n))/d
    return max(0, c-h), min(1, c+h)

for path in sorted(glob.glob('data_wacv_scenario_v5/eval_dialog/*/results.jsonl')):
    rows = {}
    for l in open(path):
        try:
            r = json.loads(l)
            if not r.get('error'): rows[(r['plan_id'], r['input_format'])] = r
        except: pass
    rows = list(rows.values())
    if len(rows) < 500: continue
    keys = set(rows and {(r['plan_id'], r['input_format']) for r in rows})
    full = {p for p, _ in keys if all((p, f) in keys for f in F)}
    rows = [r for r in rows if r['plan_id'] in full and r['evidence_pref_turn'] is not None]

    print("=" * 72); print(path.split('/')[-2], f"  n={len(rows)//2}")

    # ── ① 노출 위치(턴) 구간별 preference ──────────────────────────
    #    48턴을 6구간으로. 초반 소실이면 앞구간이 낮고,
    #    lost-in-the-middle 이면 가운데가 낮고 양끝이 높다(U자).
    for fmt in F:
        sub = [r for r in rows if r['input_format'] == fmt]
        if not sub: continue
        print(f"\n  [{fmt}]  노출 턴 구간별 preference")
        print(f"    {'구간':12s}{'n':>6}{'pref':>8}{'95% CI':>18}  {'':<22}")
        agg = defaultdict(lambda: [0, 0])
        for r in sub:
            b = min(int(r['evidence_pref_turn']) // 8, 5)
            a = agg[b]; a[0] += r['preference']; a[1] += 1
        for b in range(6):
            k, n = agg[b]
            if not n: continue
            lo, hi = wilson(k, n)
            bar = "█" * round((k/n) * 30)
            print(f"    턴 {b*8:2d}-{b*8+7:2d}   {n:6d}{k/n:8.1%}  [{lo:5.1%},{hi:5.1%}]  {bar}")

    # ── ② 선형 추세: 턴이 뒤로 갈수록 잘 맞히는가 ──────────────────
    sub = [r for r in rows if r['input_format'] == 'dialog']
    if len(sub) > 30:
        xs = [r['evidence_pref_turn'] for r in sub]
        ys = [r['preference'] for r in sub]
        n = len(xs); mx = sum(xs)/n; my = sum(ys)/n
        sxy = sum((x-mx)*(y-my) for x, y in zip(xs, ys))
        sxx = sum((x-mx)**2 for x in xs)
        syy = sum((y-my)**2 for y in ys)
        slope = sxy/sxx if sxx else 0
        r_ = sxy/math.sqrt(sxx*syy) if sxx and syy else 0
        print(f"\n  선형 추세 (dialog): 기울기 {slope*100:+.3f}%p/턴, r={r_:+.3f}")
        print(f"    → 양수면 '뒤쪽 노출이 유리'(초반 망각), 0 근처면 위치 무관")

    # ── ③ 첫/중간/마지막 에피소드 비교 ─────────────────────────────
    for fmt in F:
        sub = [r for r in rows if r['input_format'] == fmt]
        if not sub: continue
        g = defaultdict(lambda: [0, 0])
        for r in sub:
            t = r['evidence_pref_turn']
            k = "첫 에피소드(0-15)" if t < 16 else ("중간(16-31)" if t < 32 else "마지막(32-47)")
            a = g[k]; a[0] += r['preference']; a[1] += 1
        parts = [f"{k} {v[0]/v[1]:.1%}(n={v[1]})" for k, v in
                 sorted(g.items(), key=lambda x: x[0])]
        print(f"  [{fmt}] " + "  |  ".join(parts))
EOF

import json, glob, pathlib
for p in sorted(glob.glob('data_wacv_scenario_v5/eval_dialog/*/results.jsonl')):
    p = pathlib.Path(p)
    rows = [json.loads(l) for l in p.open() if l.strip()]
    keep = [r for r in rows if not r.get("error")]
    if len(rows) == len(keep):
        print(f"  {p.parent.name}: 에러 없음 ({len(rows)}행)")
        continue
    p.write_text("".join(json.dumps(r, ensure_ascii=False)+"\n" for r in keep))
    print(f"* {p.parent.name}: {len(rows)} → {len(keep)}  (에러 {len(rows)-len(keep)}건 제거)")
